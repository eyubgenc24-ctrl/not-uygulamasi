git commit -m ""  commit atmaya yarar
git log --oneline atılmış commitleri gösterir
pwd bulunduğun klasörü gösterir
cd bulunduğun klasörü değiştirir
git status stages durumunu gösterir.

git status: Hangi dosyaların değiştiğini, hazırlandığını gösterir.
git add <dosya> / git add .: Değişiklikleri commit için hazırlık alanına (staging) ekler.
git commit -m "mesaj": Hazırlanan değişiklikleri bir kayıt olarak saklar.
git diff: Henüz eklenmemiş değişikliklerin farkını gösterir.
git log: Commit geçmişini listeler (--oneline ile kısa görünüm)..
